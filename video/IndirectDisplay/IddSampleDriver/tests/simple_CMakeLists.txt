cmake_minimum_required(VERSION 3.16)

# 設置測試框架
include(CTest)
enable_testing()

# 設置編譯選項
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 創建簡化測試可執行文件
add_executable(simple_test simple_test.cpp)

# 鏈接庫
target_link_libraries(simple_test
    PRIVATE
    vddsdk
    setupapi
    advapi32
    user32
    kernel32
)

# 設置測試屬性
set_target_properties(simple_test PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
)

# 添加測試到 CTest
add_test(NAME SimpleTest COMMAND simple_test)

# 設置測試超時
set_tests_properties(SimpleTest PROPERTIES TIMEOUT 60)

# 添加測試標籤
set_tests_properties(SimpleTest PROPERTIES LABELS "simple;unit")

# 創建測試數據目錄
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/test_data)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/test_logs)

# 設置測試環境變量
set_tests_properties(SimpleTest PROPERTIES
    ENVIRONMENT "VDD_TEST_MODE=1;VDD_LOG_LEVEL=debug"
)
